GRE over IPsec 安全广域互联 —— 各设备完整配置
拓扑：AR1(总部)  Hub，AR2(上海)/AR3(广州) Spoke，AR4/5/6 模拟 ISP
封装顺序：先 GRE 封装，再 IPsec(ESP 传输模式) 保护 GRE(协议号47)

########################### AR1 北京总部 ###########################
system-view
sysname AR1-HQ

# ---- 基础接口 ----
interface GigabitEthernet0/0/0
 ip address 192.168.10.1 255.255.255.0
 quit
interface GigabitEthernet0/0/1
 ip address 203.0.113.1 255.255.255.252
 quit

# ---- 默认路由指向 ISP ----
ip route-static 0.0.0.0 0.0.0.0 203.0.113.2

# ---- GRE 隧道（上海 / 广州）----
interface Tunnel0/0/0
 ip address 10.255.12.1 255.255.255.252
 tunnel-protocol gre
 source 203.0.113.1
 destination 198.51.100.1
 gre key 120
 quit
interface Tunnel0/0/1
 ip address 10.255.13.1 255.255.255.252
 tunnel-protocol gre
 source 203.0.113.1
 destination 192.0.2.1
 gre key 130
 quit

# ---- 企业 OSPF（跑在 GRE 上）----
ospf 1 router-id 1.1.1.1
 area 0.0.0.0
  network 192.168.10.0 0.0.0.255
  network 10.255.12.0 0.0.0.3
  network 10.255.13.0 0.0.0.3
 quit

# ---- IPsec：ACL 只匹配 GRE 流量 ----
acl number 3100
 rule 5 permit gre source 203.0.113.1 0 destination 198.51.100.1 0
 quit
acl number 3101
 rule 5 permit gre source 203.0.113.1 0 destination 192.0.2.1 0
 quit

ipsec proposal GRE-TRANS
 encapsulation-mode transport
 esp authentication-algorithm sha1
 esp encryption-algorithm aes-128
 quit

ike proposal 10
 authentication-method pre-share
 authentication-algorithm sha1
 encryption-algorithm aes-cbc-128
 dh group2
 quit

ike peer SH v1
 pre-shared-key simple __PSK_SH__
 ike-proposal 10
 remote-address 198.51.100.1
 quit
ike peer GZ v1
 pre-shared-key simple __PSK_GZ__
 ike-proposal 10
 remote-address 192.0.2.1
 quit

ipsec policy GRE-IPSEC 10 isakmp
 security acl 3100
 ike-peer SH
 proposal GRE-TRANS
 quit
ipsec policy GRE-IPSEC 20 isakmp
 security acl 3101
 ike-peer GZ
 proposal GRE-TRANS
 quit

# ---- 应用到公网接口 ----
interface GigabitEthernet0/0/1
 ipsec policy GRE-IPSEC
 quit

return
save
y
